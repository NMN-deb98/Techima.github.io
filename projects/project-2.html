ماشین‌حساب پایتون

یک ماشین‌حساب ساده و سریع نوشته‌شده با پایتون؛ منطق پشت آن را ببینید یا مستقیم با نسخه‌ی زنده‌ی آن کار کنید.

0

نسخه‌ی وب این ماشین‌حساب با جاوااسکریپت شبیه‌سازی شده تا در مرورگر قابل استفاده باشد؛ منطق اصلی همان چیزی است که در کد پایتون زیر می‌بینید.

# calculator.py
def calculate(expression):
    """Safely evaluate a basic math expression."""
    allowed = "0123456789+-*/(). "
    if any(ch not in allowed for ch in expression):
        raise ValueError("invalid characters")
    return eval(expression, {"__builtins__": {}})

def main():
    print("ماشین‌حساب نیما — برای خروج q را بزنید")
    while True:
        expr = input(">> ")
        if expr.lower() == "q":
            break
        try:
            result = calculate(expr)
            print(f"= {result}")
        except Exception as e:
            print(f"خطا: {e}")

if __name__ == "__main__":
    main()
